class NumMatrix:

    def __init__(self, matrix: List[List[int]]):
        if not matrix or not matrix[0]:
            return
        n = len(matrix)
        m = len(matrix[0])
        self.prefix_sum = [[0] * m for _ in range(n)]

        self.prefix_sum[0][0] = matrix[0][0]

        for i in range(1,n):
            self.prefix_sum[i][0] = self.prefix_sum[i-1][0] + matrix[i][0] 

        for j in range(1,m):
            self.prefix_sum[0][j] = self.prefix_sum[0][j-1] + matrix[0][j]       
     
        for i in range(1,n):
            for j in range(1,m):
                self.prefix_sum[i][j] = matrix[i][j] + self.prefix_sum[i-1][j] + self.prefix_sum[i][j-1] - self.prefix_sum[i-1][j-1]
        for i in range(n):
            for j in range(m):
                print(self.prefix_sum[i][j])        

    def sumRegion(self, row1: int, col1: int, row2: int, col2: int) -> int:
        extra = 0
        if(row1!=0):
            extra +=self.prefix_sum[row1-1][col2]
        if(col1!=0):
            extra += self.prefix_sum[row2][col1-1]
        if(row1!=0 and col1!=0):
            extra -= self.prefix_sum[row1-1][col1-1]         

        return self.prefix_sum[row2][col2] - extra
