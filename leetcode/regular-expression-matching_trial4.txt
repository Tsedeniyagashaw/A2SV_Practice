class Solution:
    def isMatch(self, s: str, p: str) -> bool:
  
            if len(p) == 0:
                if len(s) == 0: return True
                return False 
            first_matched =False

            if len(s) > 0 and (p[0] == s[0] or p[0] == '.'):
                first_matched = True    

            if len(p) > 1 and p[1] == '*':
                not_take  = self.isMatch(s, p[2:])
                take = first_matched and self.isMatch(s[1:], p) 
                return take or not_take  
            else: return first_matched and self.isMatch(s[1:], p[1:])
