class Solution:
    def predictTheWinner(self, nums: List[int]) -> bool:
        def game(l,r):
            if l == r:
                return nums[l]
            diff1 = nums[l] -  game(l + 1, r)
            diff2 = nums[r] -  game(l, r - 1)
            return max(diff1,diff2)
        return game(0,len(nums)-1) >= 0
