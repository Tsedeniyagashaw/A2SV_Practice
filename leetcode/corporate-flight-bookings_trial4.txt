class Solution:
    def corpFlightBookings(self, bookings: List[List[int]], n: int) -> List[int]:
     
        output = [0] * n

        
        for start,end,seat in bookings:
            output[start - 1] += seat
            if end < n : output[end] -= seat
            
        
        for i in range(1,len(output)):
                output[i] += output[i - 1]

        return output
