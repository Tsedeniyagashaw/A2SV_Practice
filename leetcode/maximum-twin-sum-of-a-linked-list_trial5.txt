# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def pairSum(self, head: Optional[ListNode]) -> int:
        fast = head
        slow = head
        prev = None
        ans = float('-inf')
        while fast and fast.next:
            fast = fast.next.next
            half = slow.next
            slow.next = prev
            prev = slow
            slow = half

        while slow: 

            ans = max(ans, prev.val + slow.val)

            prev = prev.next 
            slow = slow.next 
        return ans
