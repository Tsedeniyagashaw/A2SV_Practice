class Solution:
    def balancedString(self, s: str) -> int:

        n = len(s)
        bal= n // 4
        count = Counter(s) 

        if all(count[c] <= bal for c in 'QWER'):
                return 0

        min_length = n
        l = 0      

        for r in range(n):
            count[s[r]] -= 1
            while all(count[c] <= bal for c in 'QWER'):
                     min_length = min(min_length, r - l + 1)
                     count[s[l]] += 1
                     l += 1
        return min_length
