class Solution:
    def pivotIndex(self, nums: List[int]) -> int:
        summ = sum(nums)

        sumLeft = 0
        sumRight = 0

        for i  in range(len(nums)):
            sumRight = summ - sumLeft - nums[i]

            if sumRight == sumLeft:
                return i
            sumLeft += nums[i]
        return -1
