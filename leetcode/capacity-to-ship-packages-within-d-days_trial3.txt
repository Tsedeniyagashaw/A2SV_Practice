class Solution:
    def shipWithinDays(self, weights: List[int], days: int) -> int:
        
        l = max(weights)

        r = sum(weights)
        while l <=  r:
            mid = (l+r) // 2
            count = 1
            tot = 0

            for w in weights:
                tot += w
                if tot > mid:
                    count += 1
                    tot = w
            if count > days:
                l = mid + 1
            else:
                r = mid - 1
        return l
