class Solution:
    def getAverages(self, nums: List[int], k: int) -> List[int]:
        res = [-1] * len(nums)

        left = 0
        window_sum = 0
        size = 2 *  k + 1

        for right in range(len(nums)):
            window_sum += nums[right]

            if right - left + 1 == size:
                res[left + k] = window_sum // size
                window_sum -= nums[left]
                left += 1
        return res
