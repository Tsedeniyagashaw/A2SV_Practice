class Solution:
    def judgeSquareSum(self, c: int) -> bool:
        i = 0
        j = int(sqrt(c))

        while i <= j:
            summ = int(i**2 + j**2)
            if summ == c:
                return True
            elif summ > c:
                j -= 1
            else:
                i += 1
        return False
