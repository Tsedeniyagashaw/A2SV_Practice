class Solution:
    def longestSemiRepetitiveSubstring(self, s: str) -> int:
        count = 0
        l = 0
        longest = 0
     
        n = len(s)
        if n <= 2:
            return n

        for r in range(1,n):
            if s[r] == s[r-1]:               
                count += 1
            while l < r and count > 1:
                if s[l] == s[l+1]:   
                   l += 1
                   count -= 1
                else:
                    l += 1            
            w = (r-l) + 1
            longest = max(longest,w)
        return longest
