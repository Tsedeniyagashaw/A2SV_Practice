class Solution:
    def shiftingLetters(self, s: str, shifts: List[List[int]]) -> str:
        
        n = len(s)
        prefix = [0] * (n + 1)
        sl =list(s)

        for l,r,d in shifts:
            move = 1 if d == 1 else -1
            prefix[l] += move
            prefix[r + 1] -= move

        for i in range(1,n):
            prefix[i] += prefix[i - 1]
        for i in range(n):
            curr = (ord(sl[i]) - ord('a') + prefix[i]) % 26
            sl[i] = chr(curr + ord('a'))
        return "".join(sl)
