class Solution:
    def searchRange(self, nums: List[int], target: int) -> List[int]:
        def search(t):
            l = 0
            r = len(nums) - 1
            while l <= r:
                mid = (l + r) // 2
                if nums[mid] < t:
                    l = mid + 1
                else:
                    r = mid - 1
            return l
        first = search(target)
        last = search(target + 1) - 1

        if first <= last:
            return [first, last]
        return [-1,-1]
