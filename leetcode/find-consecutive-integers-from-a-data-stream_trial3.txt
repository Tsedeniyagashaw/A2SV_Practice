class DataStream:
    last_element = 0
    times_seen = 0

    def __init__(self, value: int, k: int):
        self.value = value
        self.k = k
   

    def consec(self, num: int) -> bool:
        if self.last_element == num:
            self.times_seen += 1
        else:
            self.times_seen = 1
        self.last_element = num

        return self.last_element == self.value and self.times_seen >= self.k
