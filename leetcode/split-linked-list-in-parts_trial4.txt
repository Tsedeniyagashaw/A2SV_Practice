# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def splitListToParts(self, head: Optional[ListNode], k: int) -> List[Optional[ListNode]]:
        res = []
        curr = head
        n = 0
        while curr:
            n += 1
            curr = curr.next 
        parts , left = n//k, n % k
        curr = head
        prev = None

        for _ in range(k):
            res.append(curr)
            for _ in range(parts):
                if curr:
                    prev = curr
                    curr = curr.next
            if left and curr:
                prev = curr
                curr = curr.next
                left -= 1         
            if prev: prev.next = None 
        return res
