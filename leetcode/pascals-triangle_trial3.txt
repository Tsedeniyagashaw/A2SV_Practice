class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        if numRows == 0:
            return []
        if numRows == 1:
            return [[1]]

        row =[1]

        triangle = self.generate(numRows - 1)
        prev = triangle[-1]

        for i in range(len(prev) - 1):
            num = prev[i] + prev[i+1]
            row.append(num)
        row.append(1)    
        triangle.append(row)
        return triangle
