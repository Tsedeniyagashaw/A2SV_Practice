# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def swapPairs(self, head: Optional[ListNode]) -> Optional[ListNode]:
        dummy = ListNode(0)
        dummy.next = head

        curr = dummy
        while curr.next and curr.next.next:
            swap1 = curr.next
            swap2 = curr.next.next

            curr.next, swap2.next, swap1.next = swap2, swap1, swap2.next
            curr = swap1
        return dummy.next
